Websim API Documentation

WebsimSocket API - For Realtime Multiplayer

// Initialize the room connection
const room = new WebsimSocket();
await room.initialize();

// Update your client's presence
room.updatePresence({
  x: 100, 
  y: 200,
  username: "player1"
});

// Update shared room state 
room.updateRoomState({
  gameState: "started",
  score: {
    team1: 5,
    team2: 3
  }
});

// Subscribe to presence updates
room.subscribePresence((allPresence) => {
  console.log("Players:", allPresence);
});

// Subscribe to room state updates
room.subscribeRoomState((state) => {
  console.log("Game state:", state);
});

// Send events to other clients
room.send({
  type: "move",
  x: 50,
  y: 75
});

// Listen for events from other clients
room.onmessage = (event) => {
  if (event.data.type === "move") {
    movePiece(event.data.x, event.data.y);
  }
};

Collection API - For Persistent Data

// Create a new record
const post = await room.collection('post').create({
  message: "Hello world!",
  timestamp: Date.now()
});

// Get all records
const posts = room.collection('post').getList();

// Get filtered records
const userPosts = room.collection('post')
  .filter({ author: "username" })
  .getList();

// Update a record
await room.collection('post').update(post.id, {
  message: "Updated message"
});

// Delete a record
await room.collection('post').delete(post.id);

// Subscribe to changes
room.collection('post').subscribe((posts) => {
  console.log("Posts updated:", posts);
});

LLM API - For AI Functionality

// Generate text with AI
const completion = await websim.chat.completions.create({
  messages: [
    {
      role: "system",
      content: "You are a helpful assistant."
    },
    {
      role: "user",
      content: "What is the capital of France?"
    }
  ]
});

// Generate images with AI
const image = await websim.imageGen({
  prompt: "A beautiful sunset over mountains",
  aspect_ratio: "16:9"
});
