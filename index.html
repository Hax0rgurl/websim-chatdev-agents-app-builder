Websim API Documentation

WebsimSocket API - Realtime Multiplayer

Initialization & Setup

// Available globally, no import needed
const room = new WebsimSocket(); 

// Call this once to connect and load initial state
await room.initialize();

Presence (Per-Client State)

// Get all presence data (read-only)
// Automatically updated after initialize()
const allPresence = room.presence; 
// { clientId1: { x: 10, y: 20, ... }, clientId2: { ... } }

// Get your client ID
const myId = room.clientId; 

// Get info about connected peers (including yourself)
// Automatically updated after initialize()
const peers = room.peers;
// { clientId1: { username: "...", avatarUrl: "..." }, ... }
const myInfo = room.peers[room.clientId];

// Update your own presence state
room.updatePresence({
  x: 100,
  y: 200,
  health: 95,
  // any key: value pairs
});

// Subscribe to presence changes
const unsubscribePresence = room.subscribePresence((currentPresence) => {
  // This callback triggers when any client's presence changes.
  // Note: room.presence is already updated before this callback.
  console.log("Presence updated:", currentPresence);
  // Update your UI based on new presence data
});
// Call unsubscribePresence() to stop listening

// Request another client to update their presence
room.requestPresenceUpdate('client-id-to-update', {
  type: 'damage', // Custom data for the request
  amount: 10 
});

// Handle presence update requests from others
const unsubscribeRequests = room.subscribePresenceUpdateRequests((updateRequest, fromClientId) => {
  const myCurrentPresence = room.presence[room.clientId];
  if (updateRequest.type === 'damage') {
    room.updatePresence({
      ...myCurrentPresence,
      health: myCurrentPresence.health - updateRequest.amount
    });
  }
});
// Call unsubscribeRequests() to stop listening

Room State (Shared Game/App State)

// Get the current room state (read-only)
// Automatically updated after initialize()
const currentRoomState = room.roomState; 
// { score: { teamA: 5, teamB: 3 }, items: { "item1": { x: 1, y: 1 } } }

// Update the room state (merges with existing state)
room.updateRoomState({
  score: { teamA: 6 }, // Updates teamA score
  items: { 
    "item1": { x: 2, y: 2 }, // Moves item1
    "item2": { x: 5, y: 5 }  // Adds item2
  } 
});

// To remove a key, set it to null
room.updateRoomState({
  items: { "item1": null } // Removes item1
});

// Subscribe to room state changes
const unsubscribeRoomState = room.subscribeRoomState((currentState) => {
  // This callback triggers when room state changes.
  // Note: room.roomState is already updated before this callback.
  console.log("Room state updated:", currentState);
  // Update your UI based on the new state
});
// Call unsubscribeRoomState() to stop listening

Events (Ephemeral Broadcasts)

// Send an event to other clients
room.send({
  type: "chat_message", // Custom event type
  message: "Hello!",
  echo: true // (Optional) false if sender shouldn't receive it
});

// Listen for incoming events
room.onmessage = (event) => {
  const data = event.data; // { type: "...", ..., clientId: "...", username: "..." }
  switch (data.type) {
    case 'connected':
      console.log(`Client connected: ${data.username} (${data.clientId})`);
      break;
    case 'disconnected':
      console.log(`Client disconnected: ${data.username} (${data.clientId})`);
      break;
    case 'chat_message':
      console.log(`${data.username}: ${data.message}`);
      break;
    // Handle other custom events
  }
};

Collection API - Persistent Data Storage

// Access a collection (e.g., 'posts')
const postsCollection = room.collection('posts'); 
// IMPORTANT: If you change the data structure, use versioning: 
// room.collection('posts_v2')

// Create a new record (returns the created record with id, username, created_at)
let newPost = await postsCollection.create({
  message: "Hello, persistent world!",
  likes: 0
});
// newPost = { id: "...", type: "posts", username: "...", created_at: "...", message: "...", likes: 0 }

// Get a list of records (returns array, newest first)
let allPosts = postsCollection.getList(); 
// Note: May be empty initially while loading. Best to use subscribe.
// Reverse if you want oldest first: allPosts.reverse()

// Filter records (only equality checks)
let myPosts = postsCollection.filter({ username: room.peers[room.clientId]?.username }).getList();
let topLevelPosts = postsCollection.filter({ parentId: null }).getList(); // Example parent/child

// Update a record (only your own records)
await postsCollection.update(newPost.id, {
  message: "Updated message",
  likes: 1 // Update specific fields
});

// Delete a record (only your own records)
await postsCollection.delete(newPost.id);

// Subscribe to changes in a list (triggers immediately & on updates)
const unsubscribePosts = postsCollection.subscribe(function (updatedPosts) {
  console.log("Posts list updated:", updatedPosts);
  // Update your UI
});
// Call unsubscribePosts() to stop listening

// Filter then subscribe
const unsubscribeMyPosts = postsCollection.filter({ username: room.peers[room.clientId]?.username })
  .subscribe(function (myUpdatedPosts) {
    console.log("My posts updated:", myUpdatedPosts);
  });

LLM API - AI Functionality

AI Chat Completion

// Simple text generation
let conversationHistory = [
  { role: "system", content: "You are a helpful pirate assistant." },
  { role: "user", content: "What be the weather today?" }
];
const completion = await websim.chat.completions.create({
  messages: conversationHistory
});
const aiResponse = completion.content; // The AI's text response
// completion object also has role: 'assistant', content: ...

// Getting JSON output
const jsonCompletion = await websim.chat.completions.create({
  messages: [
    { role: "system", content: "Extract name and age. Respond with JSON: { name: string, age: number }" },
    { role: "user", content: "My name is Bob, I'm 30." }
  ],
  json: true // Request JSON output
});
const jsonData = JSON.parse(jsonCompletion.content); 
// jsonData = { name: "Bob", age: 30 }

// Image input
const imageAnalysis = await websim.chat.completions.create({
  messages: [
    { role: "user", content: [
        { type: "text", text: "Is this a cat? Respond JSON { isCat: boolean }" },
        { type: "image_url", image_url: { url: imageDataUrl } } // imageDataUrl from file upload or other source
      ] 
    }
  ],
  json: true
});
const analysisResult = JSON.parse(imageAnalysis.content);
