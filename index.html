Websim Design Office Developer API

LLM Chat API

// Chat completions with JSON output
const messages = [
  { role: 'system', content: 'You are a professional design office team collaborating on Websim web projects.' },
  { role: 'user', content: 'Outline the UI layout' }
];
const aiResponse = await websim.chat.completions.create({
  messages,
  json: true
});
console.log(aiResponse.content);

Image Generation API

// Generate design mockup
const image = await websim.imageGen({
  prompt: 'Modern dashboard wireframe, light theme',
  aspect_ratio: '16:9'
});
console.log(image.url);

WebsimSocket Real‑time Collaboration

// Initialize collaborative session
const room = new WebsimSocket();
await room.initialize();

// Update presence/status
room.updatePresence({ status: 'design_in_progress' });

// Subscribe to shared state
room.subscribeRoomState(state => console.log('State:', state));
room.subscribePresence(presence => console.log('Presence:', presence));

// Broadcast a custom event
room.send({
  type: 'design_update',
  details: 'Added header component',
  echo: false
});

Collection API (Persistent Records)

// Save and list design records
const record = await room.collection('design').create({
  name: 'Homepage Wireframe',
  status: 'draft'
});
const designs = room.collection('design').getList();
console.log(designs);
